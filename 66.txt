    def update_system_info(self):
        self.system_info_text.delete(1.0, tk.END)
        
        # 操作系统信息
        os_info = platform.system() + " " + platform.release() + " " + platform.version()
        self.system_info_text.insert(tk.END, f"操作系统: {os_info}\n\n")
        
        # 计算机名称和用户名
        computer_name = socket.gethostname()
        username = os.getlogin()
        self.system_info_text.insert(tk.END, f"计算机名称: {computer_name}\n")
        self.system_info_text.insert(tk.END, f"当前用户: {username}\n\n")
        
        # 处理器信息
        cpu_info = platform.processor()
        cpu_count = psutil.cpu_count(logical=False)
        cpu_logical_count = psutil.cpu_count(logical=True)
        self.system_info_text.insert(tk.END, f"处理器: {cpu_info}\n")
        self.system_info_text.insert(tk.END, f"物理核心: {cpu_count}, 逻辑核心: {cpu_logical_count}\n")
        self.system_info_text.insert(tk.END, f"当前CPU使用率: {psutil.cpu_percent(interval=1)}%\n\n")
        
        # 内存信息
        mem = psutil.virtual_memory()
        mem_total = self.format_size(mem.total)
        mem_available = self.format_size(mem.available)
        mem_used = self.format_size(mem.used)
        self.system_info_text.insert(tk.END, f"总内存: {mem_total}\n")
        self.system_info_text.insert(tk.END, f"可用内存: {mem_available}\n")
        self.system_info_text.insert(tk.END, f"已使用内存: {mem_used} ({mem.percent}%)\n\n")
        
        # 磁盘信息
        self.system_info_text.insert(tk.END, "磁盘信息:\n")
        for disk in psutil.disk_partitions():
            try:
                disk_usage = psutil.disk_usage(disk.mountpoint)
                self.system_info_text.insert(tk.END, f"  {disk.device} - {disk.mountpoint} ({disk.fstype})\n")
                self.system_info_text.insert(tk.END, f"    总容量: {self.format_size(disk_usage.total)}\n")
                self.system_info_text.insert(tk.END, f"    已使用: {self.format_size(disk_usage.used)} ({disk_usage.percent}%)\n")
                self.system_info_text.insert(tk.END, f"    可用: {self.format_size(disk_usage.free)}\n\n")
            except PermissionError:
                continue
        
        # 网络信息
        self.system_info_text.insert(tk.END, "网络信息:\n")
        net_if_addrs = psutil.net_if_addrs()
        for interface_name, interface_addresses in net_if_addrs.items():
            for address in interface_addresses:
                if str(address.family) == 'AddressFamily.AF_INET':
                    self.system_info_text.insert(tk.END, f"  {interface_name}: {address.address}\n")
        
        # 系统启动时间
        boot_time = datetime.datetime.fromtimestamp(psutil.boot_time())
        uptime = datetime.datetime.now() - boot_time
        self.system_info_text.insert(tk.END, f"\n系统启动时间: {boot_time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        self.system_info_text.insert(tk.END, f"系统运行时间: {str(uptime).split('.')[0]}\n")
        
        self.status_var.set("系统信息已更新")
    
    # 格式化大小显示
    def format_size(self, size_bytes):
        units = ['B', 'KB', 'MB', 'GB', 'TB']
        size = size_bytes
        unit_index = 0
        while size >= 1024 and unit_index < len(units) - 1:
            size /= 1024
            unit_index += 1
        return f"{size:.2f} {units[unit_index]}"

    def update_process_list(self):
        # 清空现有列表
        for item in self.process_tree.get_children():
            self.process_tree.delete(item)
        
        # 获取所有进程
        try:
            processes = []
            for proc in psutil.process_iter(['pid', 'name', 'cpu_percent', 'memory_percent', 'status', 'username']):
                try:
                    processes.append(proc.info)
                except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
                    pass
            
            # 按CPU使用率排序
            processes.sort(key=lambda x: x['cpu_percent'] or 0, reverse=True)
            
            # 添加到表格
            for proc in processes:
                self.process_tree.insert("", tk.END, values=(
                    proc['pid'],
                    proc['name'],
                    proc['cpu_percent'] or 0,
                    proc['memory_percent'] or 0,
                    proc['status'],
                    proc['username'] or "未知"
                ))
            
            self.status_var.set(f"已加载 {len(processes)} 个进程")
        except Exception as e:
            messagebox.showerror("错误", f"获取进程列表失败: {str(e)}")
            self.status_var.set("获取进程列表失败")
    
    def filter_processes(self):
        search_text = self.process_search_var.get().lower()
        # 清空现有列表
        for item in self.process_tree.get_children():
            self.process_tree.delete(item)
        
        # 筛选进程
        try:
            processes = []
            for proc in psutil.process_iter(['pid', 'name', 'cpu_percent', 'memory_percent', 'status', 'username']):
                try:
                    if search_text in proc.info['name'].lower():
                        processes.append(proc.info)
                except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
                    pass
            
            # 按CPU使用率排序
            processes.sort(key=lambda x: x['cpu_percent'] or 0, reverse=True)
            
            # 添加到表格
            for proc in processes:
                self.process_tree.insert("", tk.END, values=(
                    proc['pid'],
                    proc['name'],
                    proc['cpu_percent'] or 0,
                    proc['memory_percent'] or 0,
                    proc['status'],
                    proc['username'] or "未知"
                ))
            
            self.status_var.set(f"找到 {len(processes)} 个匹配进程")
        except Exception as e:
            messagebox.showerror("错误", f"筛选进程失败: {str(e)}")
    
    def on_process_select(self, event):
        # 启用或禁用结束进程按钮
        selected_items = self.process_tree.selection()
        if selected_items:
            self.kill_process_btn.config(state=tk.NORMAL)
        else:
            self.kill_process_btn.config(state=tk.DISABLED)
    
    def kill_selected_process(self):
        selected_items = self.process_tree.selection()
        if not selected_items:
            return
        
        for item in selected_items:
            pid = int(self.process_tree.item(item, "values")[0])
            name = self.process_tree.item(item, "values")[1]
            
            if messagebox.askyesno("确认", f"确定要结束进程 {name} (PID: {pid}) 吗?"):
                try:
                    proc = psutil.Process(pid)
                    proc.terminate()
                    # 等待进程结束
                    time.sleep(1)
                    if proc.is_running():
                        # 强制结束
                        proc.kill()
                    self.status_var.set(f"已结束进程 {name} (PID: {pid})")
                except psutil.NoSuchProcess:
                    messagebox.showinfo("提示", f"进程 {name} (PID: {pid}) 不存在")
                except psutil.AccessDenied:
                    messagebox.showerror("错误", f"没有权限结束进程 {name} (PID: {pid})")
                except Exception as e:
                    messagebox.showerror("错误", f"结束进程失败: {str(e)}")
        
        # 刷新进程列表
        self.update_process_list()

    def perform_disk_cleanup(self):
        self.cleanup_result_text.delete(1.0, tk.END)
        self.status_var.set("正在执行磁盘清理...")
        
        try:
            cleaned_size = 0
            
            # 清理系统临时文件
            if self.temp_files_var.get():
                self.cleanup_result_text.insert(tk.END, "清理系统临时文件...\n")
                self.root.update()
                
                temp_paths = [
                    os.path.join(os.environ.get("WINDIR", ""), "Temp"),
                    os.path.join(os.environ.get("USERPROFILE", ""), "AppData", "Local", "Temp")
                ]
                
                for path in temp_paths:
                    if os.path.exists(path):
                        size = self.clean_directory(path)
                        cleaned_size += size
                        self.cleanup_result_text.insert(tk.END, f"  从 {path} 清理了 {self.format_size(size)}\n")
                        self.root.update()
            
            # 清空回收站
            if self.recycle_bin_var.get() and platform.system() == "Windows":
                self.cleanup_result_text.insert(tk.END, "\n清空回收站...\n")
                self.root.update()
                
                try:
                    # 使用PowerShell命令清空回收站
                    subprocess.run(
                        ["powershell", "-Command", "Clear-RecycleBin -Force"],
                        check=True,
                        stdout=subprocess.PIPE,
                        stderr=subprocess.PIPE,
                        text=True
                    )
                    self.cleanup_result_text.insert(tk.END, "  回收站已清空\n")
                except Exception as e:
                    self.cleanup_result_text.insert(tk.END, f"  清空回收站失败: {str(e)}\n")
            
            # 清理浏览器缓存
            if self.browser_cache_var.get():
                self.cleanup_result_text.insert(tk.END, "\n清理浏览器缓存...\n")
                self.root.update()
                
                # Chrome缓存路径
                chrome_cache = os.path.join(
                    os.environ.get("USERPROFILE", ""),
                    "AppData", "Local", "Google", "Chrome", "User Data", "Default", "Cache"
                )
                if os.path.exists(chrome_cache):
                    size = self.clean_directory(chrome_cache)
                    cleaned_size += size
                    self.cleanup_result_text.insert(tk.END, f"  从Chrome缓存清理了 {self.format_size(size)}\n")
                    self.root.update()
                
                # Edge缓存路径
                edge_cache = os.path.join(
                    os.environ.get("USERPROFILE", ""),
                    "AppData", "Local", "Microsoft", "Edge", "User Data", "Default", "Cache"
                )
                if os.path.exists(edge_cache):
                    size = self.clean_directory(edge_cache)
                    cleaned_size += size
                    self.cleanup_result_text.insert(tk.END, f"  从Edge缓存清理了 {self.format_size(size)}\n")
                    self.root.update()
            
            self.cleanup_result_text.insert(tk.END, f"\n清理完成，共释放 {self.format_size(cleaned_size)} 磁盘空间")
            self.status_var.set(f"磁盘清理完成，释放 {self.format_size(cleaned_size)} 空间")
            
        except Exception as e:
            self.cleanup_result_text.insert(tk.END, f"\n清理过程中发生错误: {str(e)}")
            self.status_var.set("磁盘清理失败")
    
    def clean_directory(self, path):
        total_size = 0
        for root, dirs, files in os.walk(path):
            for file in files:
                file_path = os.path.join(root, file)
                try:
                    if os.path.isfile(file_path):
                        file_size = os.path.getsize(file_path)
                        os.remove(file_path)
                        total_size += file_size
                except Exception:
                    continue
            
            # 尝试删除空目录
            for dir in dirs:
                dir_path = os.path.join(root, dir)
                try:
                    os.rmdir(dir_path)
                except Exception:
                    continue
        
        return total_size

    def perform_ping(self):
        target = self.ping_target_var.get()
        if not target:
            messagebox.showwarning("警告", "请输入目标地址")
            return
        
        self.ping_result_text.delete(1.0, tk.END)
        self.status_var.set(f"正在Ping {target}...")
        
        try:
            # 根据操作系统构建ping命令
            param = '-n' if platform.system().lower() == 'windows' else '-c'
            command = ['ping', param, '4', target]
            
            result = subprocess.run(command, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
            self.ping_result_text.insert(tk.END, result.stdout)
            
            if result.returncode == 0:
                self.status_var.set(f"Ping {target} 成功")
            else:
                self.ping_result_text.insert(tk.END, f"\n错误: {result.stderr}")
                self.status_var.set(f"Ping {target} 失败")
                
        except Exception as e:
            self.ping_result_text.insert(tk.END, f"执行Ping时发生错误: {str(e)}")
            self.status_var.set("Ping操作失败")
    
    def perform_port_scan(self):
        target = self.port_target_var.get()
        port_range = self.port_range_var.get()
        
        if not target or not port_range:
            messagebox.showwarning("警告", "请输入目标地址和端口范围")
            return
        
        try:
            start_port, end_port = map(int, port_range.split('-'))
            if start_port < 1 or end_port > 65535 or start_port > end_port:
                raise ValueError
        except ValueError:
            messagebox.showwarning("警告", "请输入有效的端口范围 (例如: 1-100)")
            return
        
        self.port_result_text.delete(1.0, tk.END)
        self.port_result_text.insert(tk.END, f"正在扫描 {target} 的端口 {start_port}-{end_port}...\n\n")
        self.root.update()
        self.status_var.set(f"正在扫描 {target} 的端口...")
        
        open_ports = []
        
        for port in range(start_port, end_port + 1):
            try:
                with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
                    s.settimeout(0.5)
                    result = s.connect_ex((target, port))
                    if result == 0:
                        open_ports.append(port)
                        self.port_result_text.insert(tk.END, f"端口 {port} 已开放\n")
                        self.root.update()
            except Exception:
                continue
        
        if open_ports:
            self.port_result_text.insert(tk.END, f"\n扫描完成，共发现 {len(open_ports)} 个开放端口")
        else:
            self.port_result_text.insert(tk.END, "\n扫描完成，未发现开放端口")
        
        self.status_var.set(f"端口扫描完成")
    
    def update_network_info(self):
        self.network_info_text.delete(1.0, tk.END)
        
        # IP地址信息
        hostname = socket.gethostname()
        ip_address = socket.gethostbyname(hostname)
        self.network_info_text.insert(tk.END, f"主机名: {hostname}\n")
        self.network_info_text.insert(tk.END, f"本地IP: {ip_address}\n\n")
        
        # 网络接口信息
        self.network_info_text.insert(tk.END, "网络接口信息:\n")
        for interface, addrs in psutil.net_if_addrs().items():
            for addr in addrs:
                if addr.family == socket.AF_INET:
                    self.network_info_text.insert(tk.END, f"  {interface}: {addr.address}\n")
        
        # 网络使用情况
        self.network_info_text.insert(tk.END, "\n网络使用情况:\n")
        net_io = psutil.net_io_counters()
        self.network_info_text.insert(tk.END, f"  总发送: {self.format_size(net_io.bytes_sent)}\n")
        self.network_info_text.insert(tk.END, f"  总接收: {self.format_size(net_io.bytes_recv)}\n")
        
        self.status_var.set("网络信息已更新")

    def browse_search_path(self):
        path = filedialog.askdirectory()
        if path:
            self.search_path_var.set(path)
    
    def perform_file_search(self):
        filename = self.file_search_var.get()
        path = self.search_path_var.get()
        
        if not filename or not path:
            messagebox.showwarning("警告", "请输入文件名和搜索路径")
            return
        
        if not os.path.exists(path):
            messagebox.showwarning("警告", "搜索路径不存在")
            return
        
        self.file_search_result.delete(1.0, tk.END)
        self.file_search_result.insert(tk.END, f"正在 {path} 中搜索 {filename}...\n\n")
        self.root.update()
        self.status_var.set(f"正在搜索文件...")
        
        found_files = []
        
        try:
            for root, dirs, files in os.walk(path):
                for file in files:
                    if filename.lower() in file.lower():
                        file_path = os.path.join(root, file)
                        found_files.append(file_path)
                        self.file_search_result.insert(tk.END, f"{file_path}\n")
                        self.root.update()
            
            self.file_search_result.insert(tk.END, f"\n搜索完成，共找到 {len(found_files)} 个文件")
            self.status_var.set(f"文件搜索完成，找到 {len(found_files)} 个文件")
            
        except Exception as e:
            self.file_search_result.insert(tk.END, f"搜索时发生错误: {str(e)}")
            self.status_var.set("文件搜索失败")
    
    def browse_shred_file(self):
        file = filedialog.askopenfilename()
        if file:
            self.shred_file_var.set(file)
    
    def shred_file(self):
        file_path = self.shred_file_var.get()
        
        if not file_path or not os.path.exists(file_path):
            messagebox.showwarning("警告", "请选择有效的文件")
            return
        
        if not messagebox.askyesno("确认", f"确定要彻底粉碎文件 {file_path} 吗？此操作不可恢复！"):
            return
        
        self.shred_result_text.delete(1.0, tk.END)
        self.status_var.set(f"正在粉碎文件...")
        
        try:
            # 简单的文件粉碎实现：覆盖文件内容后删除
            if os.path.isfile(file_path):
                # 打开文件并写入随机数据多次
                with open(file_path, "ba+") as f:
                    length = f.tell()
                    f.seek(0)
                    f.write(os.urandom(length))
                
                # 再次覆盖
                with open(file_path, "ba+") as f:
                    f.seek(0)
                    f.write(b'\x00' * length)
                
                # 删除文件
                os.remove(file_path)
                
                self.shred_result_text.insert(tk.END, f"文件 {file_path} 已成功粉碎并删除")
                self.status_var.set("文件粉碎完成")
            else:
                self.shred_result_text.insert(tk.END, "选择的不是一个文件")
                
        except Exception as e:
            self.shred_result_text.insert(tk.END, f"粉碎文件时发生错误: {str(e)}")
            self.status_var.set("文件粉碎失败")
